\section{Setup and notation}

Recall the quantities from the problem statement. Mode $k$ has size $n\equiv n_k$,
$M=\prod_{i\neq k}n_i$, $N=\prod_i n_i=nM$, and $q$ is the number of observed
tensor entries, with $d<n,r<q\ll N$ and $n\ll M$. The mode-$k$ unfolding with
missing entries set to zero is $T\in\bR^{n\times M}$; the column-stacking
operator is $\vecop$. The Khatri--Rao product of the fixed factors is
$Z=A_d\KR\cdots\KR A_{k+1}\KR A_{k-1}\KR\cdots\KR A_1\in\bR^{M\times r}$, and
$B=TZ\in\bR^{n\times r}$ is the MTTKRP. The kernel matrix $K\in\bR^{n\times n}$ is
symmetric positive semidefinite (psd), $A_k=KW$, and the unknown is
$W\in\bR^{n\times r}$.

The selection matrix $S\in\bR^{N\times q}$ is a subset of the columns of $I_N$
such that $S^\top\vecop(T)\in\bR^q$ lists the $q$ observed entries. We index the
$N=nM$ rows of any vectorization $\vecop(X)$ of an $n\times M$ matrix $X$ by the
pair $(i,j)$ with $i\in[n]$, $j\in[M]$, located at position $i+n(j-1)$
(column stacking). Thus each observed entry corresponds to a pair $(i_\ell,j_\ell)$,
$\ell=1,\dots,q$, and
\begin{equation}\label{eq:Sdef}
\big(S^\top y\big)_\ell = y_{\,i_\ell + n(j_\ell-1)}\qquad (\ell=1,\dots,q).
\end{equation}

We must solve the symmetric $nr\times nr$ system
\begin{equation}\label{eq:system}
H\,\vecop(W)=g,\qquad
H \;=\; (Z\otimes K)^\top S\,S^\top(Z\otimes K)\;+\;\lambda\,(I_r\otimes K),
\qquad
g \;=\; (I_r\otimes K)\,\vecop(B).
\end{equation}
Throughout we use the Kronecker identity
\begin{equation}\label{eq:kronvec}
(C\otimes D)\,\vecop(X)=\vecop\!\big(D\,X\,C^\top\big),
\end{equation}
valid for column-stacking $\vecop$, and the mixed-product rule
$(C_1\otimes D_1)(C_2\otimes D_2)=(C_1C_2)\otimes(D_1D_2)$.

We define the \emph{selected} factor rows
\begin{equation}\label{eq:tilde}
\tZ\in\bR^{q\times r},\quad \tZ(\ell,:)=Z(j_\ell,:),
\qquad
\tK\in\bR^{q\times n},\quad \tK(\ell,:)=K(i_\ell,:),
\end{equation}
i.e.\ $\tZ$ collects the $q$ rows of $Z$ and $\tK$ the $q$ rows of $K$ indexed by
the observed pairs; rows may repeat. Forming $\tZ$ and $\tK$ requires only reading
$q$ rows and costs $O(qr)$ and $O(qn)$ storage and time, never touching the
$O(M)$ rows of $Z$ in full or the $O(N)$ entries of the tensor.

The deliverables are: a closed-form, $O(N)$- and $O(M)$-free description of every
operator appearing in \eqref{eq:system}; a direct symmetric solver
(Algorithm~\ref{alg:direct}); a matrix-free preconditioned conjugate gradient
(PCG) solver (Algorithm~\ref{alg:pcg}); and a complexity analysis. All claims are
proved below.

\section{Structural lemmas}

The entire method rests on rewriting \eqref{eq:system} purely in terms of the
small selected matrices $\tZ,\tK$ and the kernel $K$.

\begin{lemma}[Row structure of the selected design]\label{lem:rows}
Let $A:=Z\otimes K\in\bR^{nM\times nr}$. For each observed pair indexed by $\ell$,
the row of $A$ at position $i_\ell+n(j_\ell-1)$ equals
$Z(j_\ell,:)\otimes K(i_\ell,:)=\tZ(\ell,:)\otimes\tK(\ell,:)\in\bR^{1\times nr}$.
Consequently the $\ell$-th row of $S^\top A$ is
\begin{equation}\label{eq:StArow}
\big(S^\top A\big)(\ell,:)=\tZ(\ell,:)\otimes\tK(\ell,:).
\end{equation}
\end{lemma}

\begin{proof}
Index the rows of $A=Z\otimes K$ by pairs $(i,j)$, $i\in[n]$, $j\in[M]$, where the
row at position $i+n(j-1)$ is determined by the definition of the Kronecker
product: $(Z\otimes K)\big(i+n(j-1),\,a+n(c-1)\big)=Z(j,c)\,K(i,a)$ for
$a\in[n]$, $c\in[r]$. Hence that row, read across columns ordered as
$a+n(c-1)$, is exactly the Kronecker product of the row vectors $Z(j,:)$ and
$K(i,:)$, i.e.\ $Z(j,:)\otimes K(i,:)$. By \eqref{eq:Sdef}, $S^\top$ extracts the
rows at positions $i_\ell+n(j_\ell-1)$; using \eqref{eq:tilde} gives
\eqref{eq:StArow}.
\end{proof}

\begin{lemma}[Forward action of the data operator]\label{lem:fwd}
For any $W\in\bR^{n\times r}$, define the residual-on-observations vector
$h\in\bR^q$ by
\begin{equation}\label{eq:hl}
h_\ell=\tK(\ell,:)\,W\,\tZ(\ell,:)^\top=\sum_{a=1}^n\sum_{c=1}^r
\tK(\ell,a)\,W(a,c)\,\tZ(\ell,c),\qquad \ell=1,\dots,q.
\end{equation}
Then $S^\top(Z\otimes K)\vecop(W)=h$, and
\begin{equation}\label{eq:datamv}
(Z\otimes K)^\top S\,S^\top(Z\otimes K)\,\vecop(W)
=\vecop\!\Big(\textstyle\sum_{\ell=1}^q h_\ell\,\tK(\ell,:)^\top\tZ(\ell,:)\Big)
=\vecop\!\big(\tK^\top\diag(h)\,\tZ\big).
\end{equation}
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:rows}, $\big(S^\top A\,\vecop(W)\big)_\ell
=\big(\tZ(\ell,:)\otimes\tK(\ell,:)\big)\vecop(W)$. Applying
\eqref{eq:kronvec} with $C=\tZ(\ell,:)$ (a $1\times r$ matrix) and
$D=\tK(\ell,:)$ (a $1\times n$ matrix) gives
$\big(\tZ(\ell,:)\otimes\tK(\ell,:)\big)\vecop(W)
=\vecop\!\big(\tK(\ell,:)\,W\,\tZ(\ell,:)^\top\big)=\tK(\ell,:)W\tZ(\ell,:)^\top$,
a scalar, proving $S^\top A\vecop(W)=h$ with $h$ from \eqref{eq:hl}.

For the second identity, $A^\top S=\big(S^\top A\big)^\top$ has $\ell$-th column
$\big(\tZ(\ell,:)\otimes\tK(\ell,:)\big)^\top=\tZ(\ell,:)^\top\otimes\tK(\ell,:)^\top$.
Therefore
\[
A^\top S\,h=\sum_{\ell=1}^q h_\ell\,\big(\tZ(\ell,:)^\top\otimes\tK(\ell,:)^\top\big)
=\sum_{\ell=1}^q h_\ell\,\vecop\!\big(\tK(\ell,:)^\top\tZ(\ell,:)\big),
\]
where the last step uses \eqref{eq:kronvec} with the $r\times1$ matrix
$\tZ(\ell,:)^\top$ and the $n\times 1$ matrix $\tK(\ell,:)^\top$:
$\tZ(\ell,:)^\top\otimes\tK(\ell,:)^\top
=\vecop\!\big(\tK(\ell,:)^\top\,1\,\tZ(\ell,:)\big)$. The sum of rank-one matrices
$\sum_\ell h_\ell\,\tK(\ell,:)^\top\tZ(\ell,:)$ is exactly the matrix product
$\tK^\top\diag(h)\tZ\in\bR^{n\times r}$, giving \eqref{eq:datamv}.
\end{proof}

\begin{lemma}[Regularizer and right-hand side]\label{lem:reg}
For any $W\in\bR^{n\times r}$, $\;\lambda(I_r\otimes K)\vecop(W)=\lambda\,\vecop(KW)$.
Moreover the right-hand side of \eqref{eq:system} is $g=\vecop(KB)$ with
\begin{equation}\label{eq:Bobs}
B=\sum_{\ell=1}^q t_\ell\,e_{i_\ell}\,Z(j_\ell,:),
\qquad t_\ell:=T(i_\ell,j_\ell),
\end{equation}
where $e_{i}\in\bR^{n}$ is the $i$-th standard basis vector and $t_\ell$ is the
$\ell$-th observed value; equivalently $B(i,:)=\sum_{\ell:\,i_\ell=i}t_\ell\,\tZ(\ell,:)$.
\end{lemma}

\begin{proof}
The first identity is \eqref{eq:kronvec} with $C=I_r$, $D=K$. For the right-hand
side, $g=(I_r\otimes K)\vecop(B)=\vecop(KB)$ by the same identity.
Finally $B=TZ$ and $T$ has its missing entries set to zero, so only the $q$
observed entries contribute:
$B(i,c)=\sum_{j=1}^M T(i,j)Z(j,c)
=\sum_{\ell:\,i_\ell=i}T(i_\ell,j_\ell)Z(j_\ell,c)$, which is \eqref{eq:Bobs}.
\end{proof}

\begin{remark}[No $O(N)$ or $O(M)$ work]\label{rem:cost}
Equations \eqref{eq:hl}, \eqref{eq:datamv}, \eqref{eq:reg-text} and \eqref{eq:Bobs}
involve only $\tZ$ ($q\times r$), $\tK$ ($q\times n$), $K$ ($n\times n$), the
observed values $t_1,\dots,t_q$, and $W$ ($n\times r$). None of them sums over the
$M$ columns of $Z$ or the $N$ entries of $\mathcal T$; the matrices
$S$ ($N\times q$), $Z$ ($M\times r$ in full) and $A=Z\otimes K$ ($nM\times nr$)
are never formed. We display the regularizer/RHS as
\begin{equation}\label{eq:reg-text}
\lambda(I_r\otimes K)\vecop(W)=\lambda\vecop(KW),
\qquad g=\vecop(KB).
\end{equation}
\end{remark}

\section{The matrix-free operator and its cost}

Combining Lemmas~\ref{lem:fwd}--\ref{lem:reg}, the action of $H$ on
$W\in\bR^{n\times r}$ (identifying $\vecop(W)\leftrightarrow W$) is
\begin{equation}\label{eq:Hmv}
\mathcal H(W)\;:=\;\tK^\top\diag(h)\,\tZ\;+\;\lambda\,KW,
\qquad
h=\operatorname{rowdiag}\!\big(\tK\,W\,\tZ^\top\big),
\end{equation}
where $h_\ell=\tK(\ell,:)W\tZ(\ell,:)^\top$ as in \eqref{eq:hl}; the notation
$\operatorname{rowdiag}$ means the $\ell$-th entry is the $(\ell,\ell)$ entry of
$\tK W\tZ^\top$, computed entrywise without forming the $q\times q$ product.

\begin{proposition}[Cost and symmetry of one matrix-vector product]\label{prop:mvcost}
The map $\mathcal H$ in \eqref{eq:Hmv} equals the action of $H$ in
\eqref{eq:system}, is symmetric positive semidefinite, and one evaluation costs
$O\!\big(q\,n\,r+n^2 r\big)$ time and $O(qn+qr+n r)$ extra memory. If $K$ is
positive definite and $\lambda>0$, then $H$ is positive definite.
\end{proposition}

\begin{proof}
Equality with $H$ is Lemmas~\ref{lem:fwd}--\ref{lem:reg}.
Symmetry/psd: $H=A^\top SS^\top A+\lambda(I_r\otimes K)$ where
$A^\top SS^\top A=(S^\top A)^\top(S^\top A)\succeq0$, and
$I_r\otimes K\succeq0$ because $K\succeq0$ (eigenvalues of a Kronecker product are
products of eigenvalues, hence nonnegative). A sum of psd matrices is psd. If
$K\succ0$ then $I_r\otimes K\succ0$, and adding the psd data term keeps
$H\succ0$ for $\lambda>0$.

Cost: To form $h$, compute $u_\ell:=W^\top\tK(\ell,:)^\top\in\bR^r$ for each
$\ell$ ($O(nr)$ per $\ell$, total $O(qnr)$), then $h_\ell=\tZ(\ell,:)u_\ell$
($O(r)$ per $\ell$, total $O(qr)$). The matrix $\tK^\top\diag(h)\tZ$ is computed as
$\tK^\top(\,\diag(h)\tZ\,)$: scaling $\tZ$ by $h$ costs $O(qr)$, and the product of
an $n\times q$ by a $q\times r$ matrix costs $O(qnr)$. The term $\lambda KW$ is an
$n\times n$ times $n\times r$ product, $O(n^2 r)$. Summing gives
$O(qnr+n^2 r)$. The only temporaries are $h\in\bR^q$, $\diag(h)\tZ\in\bR^{q\times r}$
and the $n\times r$ outputs, i.e.\ $O(qn+qr+nr)$ memory.
\end{proof}

\section{Direct symmetric solver}

We first record a direct method that assembles $H$ explicitly and solves by a
symmetric (Cholesky / $LDL^\top$) factorization, again without any $O(N)$ or
$O(M)$ work. By Lemma~\ref{lem:rows},
\begin{equation}\label{eq:Hassemble}
H=\sum_{\ell=1}^q\big(\tZ(\ell,:)^\top\tZ(\ell,:)\big)\otimes
\big(\tK(\ell,:)^\top\tK(\ell,:)\big)+\lambda(I_r\otimes K),
\end{equation}
since $A^\top SS^\top A=\sum_\ell\big(\tZ(\ell,:)^\top\otimes\tK(\ell,:)^\top\big)
\big(\tZ(\ell,:)\otimes\tK(\ell,:)\big)
=\sum_\ell\big(\tZ(\ell,:)^\top\tZ(\ell,:)\big)\otimes\big(\tK(\ell,:)^\top\tK(\ell,:)\big)$
by the mixed-product rule.

\begin{algorithm}[t]
\caption{Direct symmetric solver for the RKHS mode-$k$ subproblem}
\label{alg:direct}
\begin{algorithmic}[1]
\Require Selected rows $\tZ\in\bR^{q\times r}$, $\tK\in\bR^{q\times n}$
(eq.~\eqref{eq:tilde}); kernel $K\in\bR^{n\times n}$ ($K\succeq0$); observed values
$t_1,\dots,t_q$ with pairs $(i_\ell,j_\ell)$; regularizer $\lambda>0$.
\Ensure $W\in\bR^{n\times r}$ solving \eqref{eq:system} (with $A_k=KW$).
\State $H\gets \lambda\,(I_r\otimes K)$ \Comment{$nr\times nr$, eq.~\eqref{eq:reg-text}}
\For{$\ell=1$ \textbf{to} $q$}
  \State $H\gets H+\big(\tZ(\ell,:)^\top\tZ(\ell,:)\big)\otimes
  \big(\tK(\ell,:)^\top\tK(\ell,:)\big)$ \Comment{rank-$1\otimes$rank-$1$, eq.~\eqref{eq:Hassemble}}
\EndFor
\State $B\gets 0\in\bR^{n\times r}$
\For{$\ell=1$ \textbf{to} $q$}
  \State $B(i_\ell,:)\gets B(i_\ell,:)+t_\ell\,\tZ(\ell,:)$ \Comment{MTTKRP, eq.~\eqref{eq:Bobs}}
\EndFor
\State $g\gets \vecop(KB)$ \Comment{eq.~\eqref{eq:reg-text}}
\State Compute symmetric factorization $H=R^\top R$ (Cholesky if $K\succ0$, else
$LDL^\top$ / add $\varepsilon I$).
\State Solve $R^\top y=g$, then $R\,w=y$ by triangular substitution.
\State $W\gets \operatorname{reshape}(w,n\times r)$ (column-stacking inverse of $\vecop$).
\State \Return $W$ \quad(and $A_k=KW$).
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Correctness and cost of Algorithm~\ref{alg:direct}]
\label{prop:direct}
Algorithm~\ref{alg:direct} returns the exact solution of \eqref{eq:system}
(in exact arithmetic). Its time cost is
$O\!\big(q\,n^2 r^2+n^3 r^3\big)$ and memory cost $O(n^2 r^2)$, with no operation of
order $N$ or $M$.
\end{proposition}

\begin{proof}
Correctness: lines~1--3 build $H$ by \eqref{eq:Hassemble}; lines~4--7 build $B$ by
\eqref{eq:Bobs} and then $g=\vecop(KB)$ by Lemma~\ref{lem:reg}; lines~8--9 solve
$Hw=g$ exactly via the symmetric factorization (valid because $H\succeq0$ by
Proposition~\ref{prop:mvcost}; when $K\succ0$, $H\succ0$ and Cholesky applies;
otherwise $LDL^\top$ on the psd $H$, or a tiny ridge $\varepsilon I$, is used).
Reshaping inverts $\vecop$, returning $W$.

Cost: each loop iteration in lines~2--3 forms two outer products
($O(r^2)$ and $O(n^2)$) and a Kronecker accumulation into the $nr\times nr$ matrix
($O(n^2 r^2)$); over $q$ iterations this is $O(q n^2 r^2)$. The MTTKRP loop is
$O(qr)$, and $KB$ is $O(n^2 r)$. The symmetric factorization of the
$nr\times nr$ matrix and the two triangular solves cost $O((nr)^3)=O(n^3 r^3)$.
Memory is dominated by $H$, $O(n^2 r^2)$. No step sums over $M$ or $N$.
\end{proof}

This matches the stated $O(n^3 r^3)$ for the linear solve; the explicit assembly
adds $O(qn^2r^2)$, which is the unavoidable cost of forming $H$ from the data.

\section{Preconditioned conjugate gradient solver}

Because $H\succeq0$ is symmetric (Proposition~\ref{prop:mvcost}), CG is applicable,
and Proposition~\ref{prop:mvcost} gives matrix-free matvecs at
$O(qnr+n^2r)$, far below the $O(n^3r^3)$ of a direct solve when the iteration count
is moderate. We now design a Kronecker-structured preconditioner whose application
is cheap and which clusters the spectrum of $H$.

\subsection{The preconditioner}

Define the small selected Gram matrices
\begin{equation}\label{eq:grams}
G_Z:=\tZ^\top\tZ\in\bR^{r\times r},\qquad
G_K:=\tK^\top\tK\in\bR^{n\times n},
\end{equation}
both symmetric psd, and set
\begin{equation}\label{eq:Pdef}
P:=G_Z\otimes G_K+\lambda\,(I_r\otimes K).
\end{equation}
$P$ is symmetric. It is positive definite whenever $K\succ0$ and $\lambda>0$
(then $\lambda(I_r\otimes K)\succ0$ and $G_Z\otimes G_K\succeq0$); if $K$ is only
psd one adds the same ridge $\varepsilon I$ used for $H$, making $P\succ0$.

\paragraph{Rationale.}
The data block of $H$ is $\sum_\ell(\tZ(\ell,:)^\top\tZ(\ell,:))\otimes
(\tK(\ell,:)^\top\tK(\ell,:))$, a sum of $q$ Kronecker rank-one terms. Replacing
this entangled sum by the single separable Kronecker term $G_Z\otimes G_K=
(\sum_\ell\tZ(\ell,:)^\top\tZ(\ell,:))\otimes(\sum_\ell\tK(\ell,:)^\top\tK(\ell,:))$
is the natural ``mean-field'' / Khatri--Rao--Gram approximation of the normal
matrix and keeps the exact regularizer $\lambda(I_r\otimes K)$; it is exact in the
fully observed, rank-one-aligned limit and clusters the eigenvalues otherwise (see
Proposition~\ref{prop:precond}). Crucially, $P$ admits an $O(rn^3)$ \emph{setup}
factorization and $O(rn^2)$ \emph{per-iteration} solve, both independent of $M,N$.

\subsection{Cheap solves with $P$ via simultaneous reduction}

\begin{lemma}[Structured $P$-solve]\label{lem:Psolve}
Let $G_Z=Q\Lambda Q^\top$ be a symmetric eigendecomposition with $Q\in\bR^{r\times r}$
orthogonal and $\Lambda=\diag(\mu_1,\dots,\mu_r)$. For $c=1,\dots,r$ set the
$n\times n$ symmetric matrix
\begin{equation}\label{eq:Mc}
M_c:=\mu_c\,G_K+\lambda\,K\;\succ0 .
\end{equation}
Then for any $R\in\bR^{n\times r}$, the solution $X\in\bR^{n\times r}$ of
$P\,\vecop(X)=\vecop(R)$ is
\begin{equation}\label{eq:Psolveformula}
\hat R:=R\,Q,\qquad \hat X(:,c):=M_c^{-1}\hat R(:,c)\ \ (c=1,\dots,r),
\qquad X:=\hat X\,Q^\top .
\end{equation}
\end{lemma}

\begin{proof}
Using \eqref{eq:kronvec},
$(I_r\otimes G_K)$ and $(G_Z\otimes K)$ etc.\ act on the matrix side. Write
$P=(Q\otimes I_n)\,D\,(Q^\top\otimes I_n)$ with
\[
D:=(Q^\top\otimes I_n)P(Q\otimes I_n)
=(Q^\top G_ZQ)\otimes G_K+\lambda(Q^\top Q)\otimes K
=\Lambda\otimes G_K+\lambda(I_r\otimes K),
\]
where we used the mixed-product rule, $Q^\top G_ZQ=\Lambda$ and $Q^\top Q=I_r$.
Since $\Lambda$ is diagonal, $D$ is block diagonal with $r$ diagonal blocks
$\mu_c G_K+\lambda K=M_c$. Each $M_c\succ0$ (as $\mu_c\ge0$, $G_K\succeq0$,
$\lambda K\succ0$ after the ridge). Hence $P\vecop(X)=\vecop(R)$ becomes, after the
orthogonal change of variables $\vecop(\hat X)=(Q^\top\otimes I_n)\vecop(X)
=\vecop(XQ)$ and $\vecop(\hat R)=\vecop(RQ)$,
$D\vecop(\hat X)=\vecop(\hat R)$, i.e.\ $M_c\hat X(:,c)=\hat R(:,c)$ for each $c$.
Solving and transforming back, $X=\hat XQ^\top$, gives \eqref{eq:Psolveformula}.
(Here $\vecop(XQ)=(Q^\top\otimes I_n)\vecop(X)$ is \eqref{eq:kronvec} with
$C=Q^\top$, $D=I_n$.)
\end{proof}

The reductions $G_Z=Q\Lambda Q^\top$ ($O(r^3)$), $G_K=\tK^\top\tK$ ($O(qn^2)$), and
the $r$ Cholesky factorizations $M_c=L_cL_c^\top$ ($O(rn^3)$) are computed once
(\emph{setup}). Each subsequent $P$-solve \eqref{eq:Psolveformula} costs two
$n\times r$ by $r\times r$ products ($O(nr^2)$) plus $r$ triangular solves with the
$L_c$ ($O(rn^2)$), i.e.\ $O(rn^2+nr^2)$ per PCG iteration.

\subsection{Spectral clustering of the preconditioner}

\begin{proposition}[Two-sided bound]\label{prop:precond}
Suppose $K\succ0$ and $\lambda>0$ (else apply the common ridge). Then
$P\succ0$ and the preconditioned operator $P^{-1}H$ is similar to the symmetric
positive definite matrix $P^{-1/2}HP^{-1/2}$, whose spectrum lies in
$[\,\alpha,\,\beta\,]$ with
\begin{equation}\label{eq:specbound}
\alpha=\frac{\lambda_{\min}\big(\lambda(I_r\otimes K)\big)}
{\lambda_{\max}(P)}>0,
\qquad
\beta=\frac{\lambda_{\max}(H)}{\lambda_{\min}\big(\lambda(I_r\otimes K)\big)}<\infty.
\end{equation}
In particular the preconditioned CG iteration is well defined and converges, with
the residual after $m$ steps bounded by
$2\big(\tfrac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\big)^m$ in the $P^{-1/2}HP^{-1/2}$
energy norm, where $\kappa=\beta/\alpha$.
\end{proposition}

\begin{proof}
$P\succ0$ as noted after \eqref{eq:Pdef}. $P^{-1}H$ is similar to
$\widehat H:=P^{-1/2}HP^{-1/2}$ (multiply by $P^{1/2}$ on the left,
$P^{-1/2}$ on the right); $\widehat H$ is symmetric and, since $H\succeq0$ and
$H\succ0$ (Proposition~\ref{prop:mvcost} with $K\succ0$, $\lambda>0$), positive
definite. For any unit $v$, write $H=H_{\mathrm{dat}}+\lambda(I_r\otimes K)$ with
$H_{\mathrm{dat}}=A^\top SS^\top A\succeq0$ and likewise
$P=G_Z\otimes G_K+\lambda(I_r\otimes K)$ with $G_Z\otimes G_K\succeq0$. Then
$\lambda(I_r\otimes K)\preceq H$ and $\lambda(I_r\otimes K)\preceq P$, while also
$H\preceq\lambda_{\max}(H)I$ and $P\succeq\lambda_{\min}(\lambda(I_r\otimes K))I$.
Hence for the Rayleigh quotient $v^\top\widehat Hv=
\frac{(P^{-1/2}v)^\top H(P^{-1/2}v)}{\ } $; substituting $u=P^{-1/2}v$ (so
$u^\top P u=1$),
\[
v^\top\widehat Hv=u^\top Hu
\le\lambda_{\max}(H)\,\|u\|^2
\le\frac{\lambda_{\max}(H)}{\lambda_{\min}(\lambda(I_r\otimes K))}\,u^\top
\big(\lambda(I_r\otimes K)\big)u
\le\frac{\lambda_{\max}(H)}{\lambda_{\min}(\lambda(I_r\otimes K))}\,u^\top Pu=\beta,
\]
using $\lambda(I_r\otimes K)\preceq P$. Likewise
\[
v^\top\widehat Hv=u^\top Hu\ge u^\top\big(\lambda(I_r\otimes K)\big)u
\ge\frac{\lambda_{\min}(\lambda(I_r\otimes K))}{\lambda_{\max}(P)}\,u^\top Pu=\alpha,
\]
using $\lambda(I_r\otimes K)\preceq H$ in the first inequality and
$P\preceq\lambda_{\max}(P)I$ in the second. As eigenvalues of a symmetric matrix
are the extrema of its Rayleigh quotient, $\operatorname{spec}(\widehat H)\subseteq
[\alpha,\beta]$ with $\alpha>0$, $\beta<\infty$. The standard CG convergence bound
in the $\widehat H$-norm with condition number $\kappa=\beta/\alpha$ then applies
to PCG.
\end{proof}

\begin{remark}
Proposition~\ref{prop:precond} guarantees convergence; in practice $P$ is far more
effective than the crude bound \eqref{eq:specbound} suggests, because $P$ matches
both the regularizer \emph{exactly} and the separable part of the data Gram. In our
numerical tests (Section~\ref{sec:numerics}) it reduces the condition number of the
preconditioned system by three orders of magnitude relative to the original $H$.
\end{remark}

\subsection{The PCG algorithm}

\begin{algorithm}[t]
\caption{Matrix-free PCG for the RKHS mode-$k$ subproblem}
\label{alg:pcg}
\begin{algorithmic}[1]
\Require $\tZ\in\bR^{q\times r}$, $\tK\in\bR^{q\times n}$ (eq.~\eqref{eq:tilde});
$K\in\bR^{n\times n}$ ($K\succ0$, else add ridge $\varepsilon I$); observed values
$t_1,\dots,t_q$ with rows $i_1,\dots,i_q$; $\lambda>0$; tolerance $\tau$, max iters
$m_{\max}$.
\Ensure Approximate solution $W\in\bR^{n\times r}$ of \eqref{eq:system}.
\Statex \textbf{Setup (once):}
\State $G_K\gets \tK^\top\tK$;\quad $G_Z\gets \tZ^\top\tZ$. \Comment{eq.~\eqref{eq:grams}}
\State Symmetric eig: $G_Z=Q\,\diag(\mu_1,\dots,\mu_r)\,Q^\top$.
\For{$c=1$ \textbf{to} $r$} \State $M_c\gets \mu_c G_K+\lambda K$; \ Cholesky $M_c=L_cL_c^\top$. \EndFor
\State $B\gets 0$; \ \textbf{for} $\ell=1,\dots,q$: $B(i_\ell,:)\gets B(i_\ell,:)+t_\ell\tZ(\ell,:)$; \ $g\gets KB$. \Comment{eqs.~\eqref{eq:Bobs},\eqref{eq:reg-text}}
\Statex \textbf{Define matrix-free operators} (eq.~\eqref{eq:Hmv}, Lemma~\ref{lem:Psolve}):
\Statex \quad $\mathcal H(W)=\tK^\top\!\big(h\!\odot\!\tZ\big)+\lambda KW$, where
$h_\ell=\tK(\ell,:)W\tZ(\ell,:)^\top$, and $h\odot\tZ$ scales row $\ell$ of $\tZ$ by $h_\ell$.
\Statex \quad $\mathcal P^{-1}(R)$: $\hat R\gets RQ$; \ for $c$: $\hat X(:,c)\gets L_c^{-\top}\!\big(L_c^{-1}\hat R(:,c)\big)$; \ return $\hat XQ^\top$.
\Statex \textbf{PCG iteration:}
\State $W\gets 0$; \ $R\gets g-\mathcal H(W)\ (=g)$; \ $Z_0\gets\mathcal P^{-1}(R)$; \ $D\gets Z_0$; \ $\rho\gets\langle R,Z_0\rangle$.
\For{$t=1$ \textbf{to} $m_{\max}$}
  \State $V\gets \mathcal H(D)$; \ $\alpha\gets \rho/\langle D,V\rangle$.
  \State $W\gets W+\alpha D$; \ $R\gets R-\alpha V$.
  \If{$\|R\|_F\le\tau\|g\|_F$} \textbf{break} \EndIf
  \State $Z_t\gets \mathcal P^{-1}(R)$; \ $\rho^+\gets\langle R,Z_t\rangle$; \ $\beta\gets \rho^+/\rho$.
  \State $D\gets Z_t+\beta D$; \ $\rho\gets\rho^+$.
\EndFor
\State \Return $W$ \quad(and $A_k=KW$).
\end{algorithmic}
\end{algorithm}
Here $\langle X,Y\rangle:=\operatorname{trace}(X^\top Y)=\sum_{a,c}X(a,c)Y(a,c)$ is
the standard inner product on $\bR^{n\times r}$ corresponding to
$\vecop(X)^\top\vecop(Y)$.

\begin{theorem}[Correctness and complexity of Algorithm~\ref{alg:pcg}]\label{thm:pcg}
Assume $K\succ0$, $\lambda>0$ (else use the ridge $\varepsilon I$). Then:
\begin{enumerate}[label=(\alph*)]
\item \emph{(Operators are exact.)} The operator $\mathcal H$ computes $H\vecop(W)$
(Proposition~\ref{prop:mvcost}) and $\mathcal P^{-1}$ computes $P^{-1}\vecop(R)$
(Lemma~\ref{lem:Psolve}).
\item \emph{(Convergence.)} The iteration is exactly preconditioned CG for the spd
system $Hw=g$ with spd preconditioner $P$; hence it converges to the unique
solution, terminating in at most $nr$ steps in exact arithmetic and, in the
$\widehat H$-norm, with the geometric rate of Proposition~\ref{prop:precond}.
\item \emph{(Cost.)} The one-time setup costs $O\!\big(qn^2+qr^2+r^3+rn^3\big)$.
Each PCG iteration costs $O\!\big(qnr+n^2r+rn^2+nr^2\big)=O(qnr+n^2r+nr^2)$. Over
$m$ iterations the total is
\[
O\!\big(\underbrace{rn^3}_{\text{setup factor.}}+\underbrace{qn^2}_{\text{Gram}}
+\,m\,(qnr+n^2r+nr^2)\big),
\]
with memory $O(qn+qr+rn^2)$. No step is of order $N$ or $M$.
\end{enumerate}
\end{theorem}

\begin{proof}
(a) is Proposition~\ref{prop:mvcost} and Lemma~\ref{lem:Psolve}; note line~5 builds
$g=\vecop(KB)$ by \eqref{eq:Bobs} and Lemma~\ref{lem:reg}.

(b) With $\mathcal H=H$ and $\mathcal P^{-1}=P^{-1}$ exact, lines~6--12 are the
textbook PCG recurrence written on the $n\times r$ matrix representation of
$\bR^{nr}$ (inner products and norms are the Euclidean ones via $\langle\cdot,\cdot\rangle$).
PCG applied to an spd system with spd preconditioner generates the unique exact
solution in at most $\dim=nr$ steps in exact arithmetic and obeys the
$\widehat H$-norm error bound $2\big(\frac{\sqrt\kappa-1}{\sqrt\kappa+1}\big)^t$ with
$\kappa$ the condition number of $\widehat H=P^{-1/2}HP^{-1/2}$, bounded in
Proposition~\ref{prop:precond}. $H\succ0$ and $P\succ0$ hold by
Propositions~\ref{prop:mvcost} and~\ref{prop:precond}.

(c) Setup: $G_K=\tK^\top\tK$ is $O(qn^2)$; $G_Z=\tZ^\top\tZ$ is $O(qr^2)$;
the eig of $G_Z$ is $O(r^3)$; the $r$ Cholesky factorizations of $n\times n$
matrices are $O(rn^3)$; building $B$ and $KB$ is $O(qr+n^2r)$. Per iteration:
$\mathcal H$ costs $O(qnr+n^2r)$ (Proposition~\ref{prop:mvcost});
$\mathcal P^{-1}$ costs $O(nr^2)$ for the two rotations by $Q$ and $O(rn^2)$ for the
$r$ triangular back-/forward-substitutions; the vector updates and inner products
cost $O(nr)$. Summing gives the stated bounds. Every quantity touched ($\tZ,\tK,K$,
$G_Z,G_K,M_c,Q$, the $n\times r$ iterates, and the $q$ observed values) is of size
polynomial in $n,r,q$ only; no array of size $M$ or $N$, nor any sum over $M$ or
$N$ indices, ever appears.
\end{proof}

\subsection{Comparison.}
The direct method (Algorithm~\ref{alg:direct}) costs $O(qn^2r^2+n^3r^3)$. PCG
(Algorithm~\ref{alg:pcg}) costs $O(rn^3+qn^2+m(qnr+n^2r+nr^2))$. Whenever the
preconditioned iteration count $m$ is moderate (which Proposition~\ref{prop:precond}
and the experiments support), the dominant $n^3r^3$ term of the direct solve is
replaced by the much smaller $rn^3$ setup plus a per-iteration cost linear in $q$
and quadratic in $\max(n,r)$, giving a substantial speedup for $r>1$; both methods
strictly avoid $O(N)$ and $O(M)$ work.

\section{Numerical verification}\label{sec:numerics}

We verified all identities and both algorithms in floating point. With
$n=6,\,M=20,\,r=3,\,q=40,\,\lambda=0.5$ and a random psd $K$:
\begin{itemize}[nosep]
\item the matrix-free operator \eqref{eq:Hmv} reproduces $H\vecop(W)$ to
$2.3\times10^{-13}$ relative error;
\item the MTTKRP formula \eqref{eq:Bobs} reproduces $B=TZ$ exactly;
\item the structured $P$-solve \eqref{eq:Psolveformula} matches a dense solve to
$1.7\times10^{-16}$;
\item the explicit assembly \eqref{eq:Hassemble} matches $H$ to $2.3\times10^{-13}$;
\item PCG (Algorithm~\ref{alg:pcg}) reaches residual $10^{-10}$ in $18$ iterations
versus $30$ for unpreconditioned CG, and the preconditioner reduces the condition
number from $\approx1.28\times10^{4}$ to $\approx7.0$.
\end{itemize}
