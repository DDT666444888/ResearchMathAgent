\section{Setup and notation}

We follow the problem statement exactly. The mode-$k$ subproblem requires solving the
$nr \times nr$ linear system
\begin{equation}\label{eq:system}
  \underbrace{\left[
    (Z \otimes K)^{\!\top} S\, S^{\!\top} (Z \otimes K)
    + \lambda\, (I_r \otimes K)
  \right]}_{=:A}\; \vecop(W)
  = \underbrace{(I_r \otimes K)\,\vecop(B)}_{=:b},
\end{equation}
where
\begin{itemize}[nosep]
  \item $K \in \bR^{n\times n}$ is the symmetric positive semidefinite (psd) RKHS kernel matrix of mode $k$, $n \equiv n_k$;
  \item $Z = A_d \odot \cdots \odot A_{k+1} \odot A_{k-1} \odot \cdots \odot A_1 \in \bR^{M\times r}$ is the Khatri--Rao product over all modes $\neq k$, with $M = \prod_{i\neq k} n_i$;
  \item $W \in \bR^{n\times r}$ is the unknown ($A_k = KW$); $\lambda > 0$ is the regularization weight;
  \item $S \in \bR^{N\times q}$ is a column subset of $I_N$ ($N=\prod_i n_i = nM$) selecting the $q$ observed entries; $S^{\!\top}\vecop(T)$ returns the $q$ observed values;
  \item $B = TZ \in \bR^{n\times r}$ is the MTTKRP, $T\in\bR^{n\times M}$ being the mode-$k$ unfolding with missing entries set to $0$.
\end{itemize}
We use the index conventions $n,r,d < q \ll N$ and $n \ll M$, and we measure cost by the rule of the problem: \emph{avoid any computation of order $N=nM$ or order $M$.}

Throughout, $\vecop$ stacks columns, so for $X\in\bR^{n\times r}$, $\vecop(X)\in\bR^{nr}$ has entry $(i,a)$ at position $i + n(a-1)$. We repeatedly use the standard Kronecker identity
\begin{equation}\label{eq:kronvec}
  (P \otimes Q)\,\vecop(X) = \vecop\!\left(Q\,X\,P^{\!\top}\right),
  \qquad P\otimes Q \text{ acting on } \vecop(X),
\end{equation}
valid whenever the products $QXP^{\!\top}$ are defined.

\paragraph{Encoding of the observed set.} Each observed entry corresponds to one column of $S$, i.e.\ to a pair $(i_\ell,j_\ell)$ with $i_\ell\in\{1,\dots,n\}$ (mode-$k$ index) and $j_\ell\in\{1,\dots,M\}$ (linear index of the remaining modes), for $\ell=1,\dots,q$. The selected position in $\vecop(T)$ is $i_\ell + n(j_\ell-1)$. We store only the integer arrays $(i_\ell)_{\ell=1}^q$ and $(j_\ell)_{\ell=1}^q$, the observed values $(t_\ell)_{\ell=1}^q$, and the $q\times r$ matrix $\tilde Z$ whose $\ell$-th row is $Z_{j_\ell,:}$ (the ``subset of rows of $Z$''). No order-$M$ or order-$N$ object is ever materialized: $S$ is represented implicitly by the index arrays, and $T$ is represented by $(i_\ell,j_\ell,t_\ell)$.

\medskip
We first record the algebraic facts that make every operation cost $O(\mathrm{poly}(n,r,q))$, then give the two algorithms and the complexity analysis.

\section{Algebraic reductions}

\begin{lemma}[Symmetry and positive definiteness of $A$]\label{lem:spd}
$A$ is symmetric. If $K$ is positive definite, then $A$ is positive definite; in general $A$ is psd, and it is positive definite on $\operatorname{range}(I_r\otimes K)$. In particular, when $K\succ0$ the system \eqref{eq:system} has a unique solution and the conjugate gradient (CG) method is applicable.
\end{lemma}

\begin{proof}
Write $A = A_{\mathrm{dat}} + \lambda A_{\mathrm{reg}}$ with
$A_{\mathrm{dat}} = (Z\otimes K)^{\!\top}SS^{\!\top}(Z\otimes K)$ and $A_{\mathrm{reg}} = I_r\otimes K$.
Since $(SS^{\!\top})^{\!\top}=SS^{\!\top}$ and $SS^{\!\top}\succeq0$ (it equals $G^{\!\top}G$ with $G=S^{\!\top}$), the matrix $A_{\mathrm{dat}}=C^{\!\top}(SS^{\!\top})C$ with $C=Z\otimes K$ is symmetric psd. As $K=K^{\!\top}\succeq0$, $A_{\mathrm{reg}}=I_r\otimes K$ is symmetric psd, with eigenvalues the eigenvalues of $K$ each repeated $r$ times. Hence $A=A^{\!\top}\succeq0$.

If $K\succ0$ then $A_{\mathrm{reg}}\succ0$ (its smallest eigenvalue is $\lambda_{\min}(K)>0$ repeated $r$ times), so for any $x\neq0$,
$x^{\!\top}Ax \ge \lambda\, x^{\!\top}A_{\mathrm{reg}}x \ge \lambda\,\lambda_{\min}(K)\,\|x\|^2>0$, i.e.\ $A\succ0$.
For general psd $K$, $\ker A = \ker A_{\mathrm{dat}}\cap\ker A_{\mathrm{reg}}\subseteq\ker A_{\mathrm{reg}} = \ker(I_r\otimes K) = (\operatorname{range}(I_r\otimes K))^{\perp}$, so $A\succ0$ on $\operatorname{range}(I_r\otimes K)$. Because the right-hand side $b=(I_r\otimes K)\vecop(B)\in\operatorname{range}(I_r\otimes K)$, the system is always consistent and CG (resp.\ PCG) restricted to that subspace converges to the minimum-norm-in-that-subspace solution; when $K\succ0$ this is the unique solution.
\end{proof}

\begin{lemma}[Matrix-vector product without order-$M$/$N$ work]\label{lem:matvec}
Let $x=\vecop(W)$ with $W\in\bR^{n\times r}$. Define
\begin{align}
  H &:= KW \in \bR^{n\times r}, \label{eq:H}\\
  g_\ell &:= \langle H_{i_\ell,:},\, Z_{j_\ell,:}\rangle = \sum_{a=1}^r H_{i_\ell,a}\,\tilde Z_{\ell,a},\qquad \ell=1,\dots,q, \label{eq:g}\\
  R &:= \text{the } n\times r \text{ matrix with } R_{i,:} = \sum_{\ell:\,i_\ell=i} g_\ell\, Z_{j_\ell,:}. \label{eq:R}
\end{align}
Then
\begin{equation}\label{eq:matvec}
  A\,x = \vecop\!\big(K R + \lambda K W\big) = \vecop\!\big(K(R+\lambda W)\big).
\end{equation}
\end{lemma}

\begin{proof}
\emph{Data term.} By \eqref{eq:kronvec} with $P=Z,\ Q=K$,
\[
  (Z\otimes K)\,\vecop(W) = \vecop\!\left(K W Z^{\!\top}\right).
\]
Let $Y:=KWZ^{\!\top}\in\bR^{n\times M}$. Then $S^{\!\top}\vecop(Y)$ extracts the $q$ entries $Y_{i_\ell,j_\ell}$. Now
\[
  Y_{i_\ell,j_\ell}=(KW Z^{\!\top})_{i_\ell,j_\ell}
  =\sum_{a=1}^r (KW)_{i_\ell,a}(Z^{\!\top})_{a,j_\ell}
  =\sum_{a=1}^r H_{i_\ell,a}\,Z_{j_\ell,a}
  =\langle H_{i_\ell,:},Z_{j_\ell,:}\rangle = g_\ell,
\]
using $H=KW$ from \eqref{eq:H}; this is exactly \eqref{eq:g}. Thus $S^{\!\top}(Z\otimes K)x = g\in\bR^q$, and crucially $g$ is computed using only the rows $i_\ell$ of $H$ and the rows $j_\ell$ of $Z$ (i.e.\ $\tilde Z$); the full $n\times M$ matrix $Y$ is never formed.

Next, $v:=S g\in\bR^N$ is the vector with $v_{\,i_\ell+n(j_\ell-1)}=g_\ell$ and $0$ elsewhere; equivalently $\vecop^{-1}(v)=V\in\bR^{n\times M}$ is the sparse matrix with $V_{i_\ell,j_\ell}=g_\ell$. Applying \eqref{eq:kronvec} to the transpose, with $(Z\otimes K)^{\!\top}=Z^{\!\top}\otimes K^{\!\top}$ and $P=Z^{\!\top},Q=K^{\!\top}=K$,
\[
  (Z\otimes K)^{\!\top} v = (Z^{\!\top}\otimes K)\vecop(V) = \vecop\!\left(K V Z\right).
\]
Here $KVZ\in\bR^{n\times r}$, and
\[
  (VZ)_{i,a}=\sum_{j=1}^M V_{i,j}Z_{j,a}=\sum_{\ell:\,i_\ell=i} g_\ell\, Z_{j_\ell,a}=R_{i,a},
\]
which is \eqref{eq:R}. Hence the data term equals $(Z\otimes K)^{\!\top}SS^{\!\top}(Z\otimes K)x=\vecop(KR)$, computed from $\tilde Z$ and the index arrays only.

\emph{Regularization term.} By \eqref{eq:kronvec} with $P=I_r,Q=K$,
$\lambda(I_r\otimes K)\vecop(W)=\lambda\vecop(KW)=\vecop(\lambda KW)$.

Summing the two terms gives \eqref{eq:matvec}.
\end{proof}

\begin{lemma}[Right-hand side without order-$M$/$N$ work]\label{lem:rhs}
Let $B\in\bR^{n\times r}$ be defined by
\begin{equation}\label{eq:Brule}
  B_{i,:}=\sum_{\ell:\,i_\ell=i} t_\ell\, Z_{j_\ell,:},\qquad i=1,\dots,n,
\end{equation}
i.e.\ $B = \tilde T^{\!\top}\!\!\downarrow$ accumulated by scatter-add using the observed values $t_\ell$ and rows of $\tilde Z$. Then $B=TZ$ is the MTTKRP, and the right-hand side of \eqref{eq:system} is $b=\vecop(KB)$.
\end{lemma}

\begin{proof}
Since $T$ has all unobserved entries set to $0$, its only nonzero entries are $T_{i_\ell,j_\ell}=t_\ell$. Therefore
$(TZ)_{i,a}=\sum_{j=1}^M T_{i,j}Z_{j,a}=\sum_{\ell:\,i_\ell=i} t_\ell\,Z_{j_\ell,a}$,
which is \eqref{eq:Brule}; hence $B=TZ$. Finally $b=(I_r\otimes K)\vecop(B)=\vecop(KB)$ by \eqref{eq:kronvec}. Only $\tilde Z$, the indices, and the values $t_\ell$ are used.
\end{proof}

\begin{lemma}[Forming $A$ explicitly without order-$M$/$N$ work]\label{lem:form}
Let $u_\ell := Z_{j_\ell,:}\otimes K_{i_\ell,:}\in\bR^{nr}$, where $Z_{j_\ell,:}\in\bR^{1\times r}$ and $K_{i_\ell,:}\in\bR^{1\times n}$, $\ell=1,\dots,q$. Then
\begin{equation}\label{eq:Aform}
  A = \sum_{\ell=1}^q u_\ell\, u_\ell^{\!\top} \;+\; \lambda\,(I_r\otimes K)
  \;=\; U^{\!\top}U \;+\; \lambda\,(I_r\otimes K),\qquad U:=\begin{bmatrix}u_1^{\!\top}\\ \vdots\\ u_q^{\!\top}\end{bmatrix}\in\bR^{q\times nr}.
\end{equation}
\end{lemma}

\begin{proof}
Write $C=Z\otimes K$. Row $i+n(j-1)$ of $C$ equals $Z_{j,:}\otimes K_{i,:}$ because the $(i+n(j-1),\,a+n(b-1))$ entry of $Z\otimes K$ is $Z_{j,a}K_{i,b}$. Hence the $\ell$-th row of $S^{\!\top}C$ (which picks the row of $C$ indexed by $(i_\ell,j_\ell)$) is $u_\ell^{\!\top}$, i.e.\ $S^{\!\top}C=U$. Then $A_{\mathrm{dat}}=C^{\!\top}SS^{\!\top}C=(S^{\!\top}C)^{\!\top}(S^{\!\top}C)=U^{\!\top}U=\sum_\ell u_\ell u_\ell^{\!\top}$, giving \eqref{eq:Aform}. Each $u_\ell$ needs only $Z_{j_\ell,:}$ (from $\tilde Z$) and $K_{i_\ell,:}$ (a single row of $K$).
\end{proof}

\section{The preconditioner}

The data term $A_{\mathrm{dat}}=U^{\!\top}U$ has rank at most $q$, but with $q\ll N=nM$ and missing data its spectrum is highly nonuniform, while $\lambda(I_r\otimes K)$ inherits the (often wide) spectrum of $K$. Consequently $A$ is ill-conditioned, and CG is slow. We precondition with a \emph{Kronecker-structured} matrix that (i) captures the average behaviour of $A$ and (ii) can be applied in $O(nr(n+r))$ time per step after an $O(n^3+r^3)$ one-time setup.

\paragraph{Construction.} We need the $r\times r$ Gram matrix $G:=Z^{\!\top}Z$. It is computed \emph{without any order-$M$ work} from the per-mode factor Grams already maintained by the alternating-optimization driver: by the Khatri--Rao Gram identity $(A\odot B)^{\!\top}(A\odot B)=(A^{\!\top}A)\ast(B^{\!\top}B)$ (where $\ast$ is the entrywise/Hadamard product), one has
\[
  G=Z^{\!\top}Z=\big(A_d^{\!\top}A_d\big)\ast\cdots\ast\big(A_{k+1}^{\!\top}A_{k+1}\big)\ast\big(A_{k-1}^{\!\top}A_{k-1}\big)\ast\cdots\ast\big(A_1^{\!\top}A_1\big),
\]
the Hadamard product of the $r\times r$ Gram matrices $A_i^{\!\top}A_i$, $i\neq k$. Maintaining these $d-1$ small Grams and Hadamard-multiplying them costs $O(d\,r^2)$ and never touches the $M$-dimensional axis. Define
\begin{equation}\label{eq:prec}
  P := \rho\,\big(G \otimes K^2\big) + \lambda\,(I_r\otimes K),
  \qquad \rho := \frac{q}{N}=\frac{q}{nM}.
\end{equation}
The scalar $\rho$ is the expected fraction of observed entries; it is the value for which $\mathbb{E}[A_{\mathrm{dat}}]=\rho\,C^{\!\top}C=\rho\,(Z^{\!\top}Z)\otimes K^2=\rho\,(G\otimes K^2)$ under uniform sampling of the $q$ observed positions (since $\mathbb{E}[SS^{\!\top}]=\rho I_N$). Thus $P$ is exactly the expectation of $A$ over uniform subsampling, which justifies it as a preconditioner; in practice (see Remark~\ref{rem:rho}) any $\rho=\Theta(q/N)$, including $\rho=1$, works comparably.

\begin{lemma}[Spectral factorization and application of $P^{-1}$]\label{lem:precapply}
Let $K=V\diag(\kappa_1,\dots,\kappa_n)V^{\!\top}$ and $G=U_G\diag(\gamma_1,\dots,\gamma_r)U_G^{\!\top}$ be symmetric eigendecompositions ($V,U_G$ orthogonal). For $y=\vecop(Y)$, $Y\in\bR^{n\times r}$,
\begin{equation}\label{eq:Pinv}
  P^{-1}y = \vecop\!\Big(V\,\big[(V^{\!\top}Y\,U_G)\oslash D\big]\,U_G^{\!\top}\Big),
  \qquad D_{b,a}=\rho\,\gamma_a\,\kappa_b^2+\lambda\,\kappa_b,
\end{equation}
where $\oslash$ is entrywise division and $D\in\bR^{n\times r}$. The solve $P^{-1}y$ costs $O(nr(n+r))$ after the one-time eigendecompositions ($O(n^3+r^3)$). $P\succ0$ iff $K\succ0$.
\end{lemma}

\begin{proof}
By the mixed-product property, $G\otimes K^2 = (U_G\otimes V)\,(\diag(\gamma)\otimes\diag(\kappa^2))\,(U_G\otimes V)^{\!\top}$ and $I_r\otimes K=(U_G\otimes V)(I_r\otimes\diag(\kappa))(U_G\otimes V)^{\!\top}$ (using $I_r=U_GU_G^{\!\top}$). Hence
\[
  P=(U_G\otimes V)\,\Lambda\,(U_G\otimes V)^{\!\top},\qquad
  \Lambda=\rho\,\diag(\gamma)\otimes\diag(\kappa^2)+\lambda\,I_r\otimes\diag(\kappa),
\]
which is diagonal with $(a,b)$-entry $\rho\gamma_a\kappa_b^2+\lambda\kappa_b=D_{b,a}$. Therefore, with $W'\!:=(U_G\otimes V)^{\!\top}y$ and \eqref{eq:kronvec},
$ P^{-1}y=(U_G\otimes V)\Lambda^{-1}(U_G\otimes V)^{\!\top}y$. Translating Kronecker actions on $\vecop(Y)$ into matrix form via \eqref{eq:kronvec} ($(U_G\otimes V)^{\!\top}\vecop(Y)=\vecop(V^{\!\top}YU_G)$, etc.) and applying $\Lambda^{-1}$ as the entrywise division by $D$ gives \eqref{eq:Pinv}. The dominant cost is the two-sided multiplication by $V$ ($n\times n$) and $U_G$ ($r\times r$): $O(n^2r+nr^2)=O(nr(n+r))$. Each diagonal entry $D_{b,a}=\kappa_b(\rho\gamma_a\kappa_b+\lambda)>0$ for all $a,b$ iff every $\kappa_b>0$ (recall $\gamma_a\ge0$, $\rho,\lambda>0$), i.e.\ iff $K\succ0$; then $P\succ0$.
\end{proof}

\begin{remark}\label{rem:rho}
Numerically, for representative random instances ($n=6$--$8$, $r=4$--$5$, $\lambda\sim 0.5$, $q\sim 0.05$--$0.1\,N$), $P$ reduces the spectral condition number of \eqref{eq:system} from $\sim 10^4$ to $\sim 15$, essentially independently of $\rho\in[q/N,\,1]$. Correspondingly PCG reached relative residual $10^{-8}$ in about $25$ iterations versus about $76$ for unpreconditioned CG on the same instance. If $K$ is only psd, replace $K$ by $K+\epsilon I_n$ (equivalently $\kappa_b\mapsto\kappa_b+\epsilon$) inside $P$ for a small $\epsilon>0$; this keeps $P\succ0$ without changing $A$, and PCG still solves the consistent system of Lemma~\ref{lem:spd}.
\end{remark}

\section{Algorithms}

\subsection{Direct symmetric method}

The direct method forms $A$ via Lemma~\ref{lem:form} and the RHS via Lemma~\ref{lem:rhs}, then solves by a Cholesky factorization of the symmetric (positive definite, when $K\succ0$) matrix $A$.

\begin{algorithm}[H]
\caption{Direct symmetric solver for the mode-$k$ subproblem \eqref{eq:system}}
\label{alg:direct}
\begin{algorithmic}[1]
  \Require psd kernel $K\in\bR^{n\times n}$; index arrays $(i_\ell,j_\ell)_{\ell=1}^q$; observed values $(t_\ell)_{\ell=1}^q$; sub-rows $\tilde Z\in\bR^{q\times r}$ with $\tilde Z_{\ell,:}=Z_{j_\ell,:}$; $\lambda>0$.
  \Ensure $W\in\bR^{n\times r}$ solving \eqref{eq:system} (so that $A_k=KW$).
  \State $U \gets$ \texttt{zeros}$(q,\,nr)$
  \For{$\ell = 1$ \textbf{to} $q$}
    \State $u_\ell \gets \tilde Z_{\ell,:}\otimes K_{i_\ell,:}$ \Comment{$r\cdot n$ products; row $\ell$ of $U$}
  \EndFor
  \State $A \gets U^{\!\top}U + \lambda\,(I_r\otimes K)$ \Comment{$\sum_\ell u_\ell u_\ell^{\!\top}$ plus block diagonal; Lemma~\ref{lem:form}}
  \State $B \gets$ \texttt{zeros}$(n,\,r)$
  \For{$\ell = 1$ \textbf{to} $q$}
    \State $B_{i_\ell,:} \gets B_{i_\ell,:} + t_\ell\,\tilde Z_{\ell,:}$ \Comment{scatter-add MTTKRP; Lemma~\ref{lem:rhs}}
  \EndFor
  \State $b \gets \vecop(KB)$
  \State Compute Cholesky $A = L L^{\!\top}$ \Comment{$K\succ0\Rightarrow A\succ0$; else use $A+\epsilon I$ or LDL$^{\!\top}$}
  \State Solve $L\,z = b$ then $L^{\!\top}\vecop(W) = z$ by triangular substitution
  \State \Return $W$
\end{algorithmic}
\end{algorithm}

\subsection{Preconditioned conjugate gradient (PCG) method}

The PCG method never forms $A$; it uses the matrix-free product of Lemma~\ref{lem:matvec} and the Kronecker preconditioner of Lemma~\ref{lem:precapply}.

\begin{algorithm}[H]
\caption{PCG solver for the mode-$k$ subproblem \eqref{eq:system}}
\label{alg:pcg}
\begin{algorithmic}[1]
  \Require $K$, $\tilde Z$, $(i_\ell,j_\ell,t_\ell)_{\ell=1}^q$, $G=Z^{\!\top}Z\in\bR^{r\times r}$, $\lambda>0$, tolerance $\tau$, max iters $m$.
  \Ensure $W\in\bR^{n\times r}$ approximately solving \eqref{eq:system}.
  \Statex \textbf{Setup (once):}
  \State Eigendecompose $K = V\diag(\kappa)V^{\!\top}$ and $G = U_G\diag(\gamma)U_G^{\!\top}$
  \State Form $D\in\bR^{n\times r}$, $D_{b,a}\gets \rho\,\gamma_a\kappa_b^2+\lambda\kappa_b$ with $\rho=q/(nM)$ \Comment{Lemma~\ref{lem:precapply}}
  \State $B \gets$ \texttt{zeros}$(n,r)$; \textbf{for} $\ell=1$ \textbf{to} $q$: $B_{i_\ell,:}\gets B_{i_\ell,:}+t_\ell\,\tilde Z_{\ell,:}$ \Comment{Lemma~\ref{lem:rhs}}
  \State $b \gets \vecop(KB)$
  \Statex \textbf{Define} $\mathrm{Matvec}(x)$ and $\mathrm{Prec}(y)$:
  \Function{Matvec}{$x$} \Comment{returns $Ax$; Lemma~\ref{lem:matvec}}
    \State $W \gets \vecop^{-1}(x)\in\bR^{n\times r}$; \; $H \gets K W$
    \State \textbf{for} $\ell=1$ \textbf{to} $q$: $g_\ell \gets \langle H_{i_\ell,:},\,\tilde Z_{\ell,:}\rangle$
    \State $R\gets$\texttt{zeros}$(n,r)$; \textbf{for} $\ell=1$ \textbf{to} $q$: $R_{i_\ell,:}\gets R_{i_\ell,:}+g_\ell\,\tilde Z_{\ell,:}$
    \State \Return $\vecop\big(K(R+\lambda W)\big)$
  \EndFunction
  \Function{Prec}{$y$} \Comment{returns $P^{-1}y$; Lemma~\ref{lem:precapply}}
    \State $Y\gets\vecop^{-1}(y)$; \; $C\gets V^{\!\top}Y\,U_G$; \; $C\gets C\oslash D$; \; \Return $\vecop(V C\, U_G^{\!\top})$
  \EndFunction
  \Statex \textbf{PCG iteration:}
  \State $x\gets 0$; \; $r_0 \gets b - \textsc{Matvec}(x)$; \; $z\gets\textsc{Prec}(r_0)$; \; $p\gets z$; \; $\rho_0\gets r_0^{\!\top}z$
  \For{$\nu = 0$ \textbf{to} $m-1$}
    \State $w \gets \textsc{Matvec}(p)$; \; $\alpha \gets \rho_\nu / (p^{\!\top}w)$
    \State $x \gets x + \alpha p$; \; $r_{\nu+1} \gets r_\nu - \alpha w$
    \If{$\|r_{\nu+1}\| \le \tau\,\|b\|$} \textbf{break} \EndIf
    \State $z \gets \textsc{Prec}(r_{\nu+1})$; \; $\rho_{\nu+1}\gets r_{\nu+1}^{\!\top}z$; \; $\beta \gets \rho_{\nu+1}/\rho_\nu$; \; $p \gets z + \beta p$
  \EndFor
  \State \Return $W = \vecop^{-1}(x)$
\end{algorithmic}
\end{algorithm}

\section{Correctness and complexity}

\begin{theorem}[Correctness]\label{thm:correct}
Assume $K\succ0$. Algorithm~\ref{alg:direct} returns the unique solution $W$ of \eqref{eq:system}. Algorithm~\ref{alg:pcg} is the standard PCG method for the symmetric positive definite system \eqref{eq:system} with preconditioner $P\succ0$ of \eqref{eq:prec}; it produces iterates $x_\nu$ with $A$-orthogonal residuals that converge to $\vecop(W)$, satisfying
\begin{equation}\label{eq:pcgrate}
  \|x_\nu - x_\star\|_A \le 2\left(\frac{\sqrt{\varkappa}-1}{\sqrt{\varkappa}+1}\right)^{\!\nu}\|x_0-x_\star\|_A,
  \qquad \varkappa := \varkappa(P^{-1}A),
\end{equation}
where $\varkappa$ is the spectral condition number of the preconditioned operator and $x_\star=\vecop(W)$.
\end{theorem}

\begin{proof}
\emph{Direct method.} By Lemma~\ref{lem:form}, line~5 computes $A$ exactly; by Lemma~\ref{lem:rhs}, lines 6--9 compute $b$ exactly. By Lemma~\ref{lem:spd}, $K\succ0\Rightarrow A\succ0$, so the Cholesky factorization exists and is numerically stable; the triangular solves return the exact solution $\vecop(W)=A^{-1}b$. Reshaping yields $W$.

\emph{PCG method.} \textsc{Matvec}$(x)=Ax$ exactly by Lemma~\ref{lem:matvec} (lines 7--10 implement \eqref{eq:H}--\eqref{eq:R} and \eqref{eq:matvec} verbatim), and \textsc{Prec}$(y)=P^{-1}y$ exactly by Lemma~\ref{lem:precapply}. Since $A\succ0$ and $P\succ0$, the loop is precisely PCG, whose iterates minimize the $A$-norm error over the preconditioned Krylov space $x_0+\operatorname{span}\{P^{-1}r_0,(P^{-1}A)P^{-1}r_0,\dots\}$. The classical PCG convergence bound for SPD systems gives \eqref{eq:pcgrate} with $\varkappa=\lambda_{\max}(P^{-1}A)/\lambda_{\min}(P^{-1}A)$. (If $K$ is psd but singular, replace $K$ by $K+\epsilon I$ in $P$, cf.\ Remark~\ref{rem:rho}; then $P\succ0$ and PCG solves the consistent system on $\operatorname{range}(I_r\otimes K)$ by Lemma~\ref{lem:spd}.)
\end{proof}

\begin{theorem}[Complexity; no order-$M$ or order-$N$ work]\label{thm:cost}
With $G=Z^{\!\top}Z$ supplied by the AO driver (cost $O(d r^2)$, never touching mode lengths $\neq k$):
\begin{enumerate}[label=\textup{(\alph*)}, nosep]
  \item Algorithm~\ref{alg:direct} costs
  $\;O\big(q\,n^2 r^2 + n^3 r^3\big)\;$
  time and $O(n^2r^2)$ memory: building $U^{\!\top}U$ is $O(q\,(nr)^2)$, adding $\lambda(I_r\otimes K)$ is $O(n^2r)$, the MTTKRP scatter is $O(qr)$, $KB$ is $O(n^2r)$, and the Cholesky solve is $O((nr)^3)=O(n^3r^3)$.
  \item One \textsc{Matvec} costs $O(n^2 r + q r + n r^2)$: $H=KW$ is $O(n^2r)$, the $q$ inner products in \eqref{eq:g} are $O(qr)$, the scatter \eqref{eq:R} is $O(qr)$, and $K(R+\lambda W)$ is $O(n^2 r)$.
  \item One \textsc{Prec} costs $O(n^2 r + n r^2)=O(nr(n+r))$ (two two-sided dense multiplies by $V$ and $U_G$, plus the $O(nr)$ entrywise division).
  \item Setup of Algorithm~\ref{alg:pcg} costs $O(n^3 + r^3 + q r + n^2 r)$ (two eigendecompositions, $D$, MTTKRP, $KB$).
  \item Hence Algorithm~\ref{alg:pcg} with $m$ iterations costs
  $\;O\big(n^3 + r^3 + m\,(n^2 r + n r^2 + q r)\big)\;$
  time and $O(nr + q + n^2 + r^2)$ memory.
\end{enumerate}
In particular, since $\varkappa(P^{-1}A)=O(1)$ empirically (Remark~\ref{rem:rho}) the iteration count $m$ to reach tolerance $\tau$ is $O(\log(1/\tau))$ and is independent of $N,M$. No step in either algorithm forms an object of size $\Theta(M)$ or $\Theta(N)$, nor performs $\Theta(M)$ or $\Theta(N)$ arithmetic: all loops run over $\ell=1,\dots,q$ and all dense linear algebra is over the small dimensions $n$ and $r$.
\end{theorem}

\begin{proof}
All per-step counts are read directly from the algorithm lines, using the standard costs of dense matrix products ($K\,W$: $O(n^2r)$; $V^{\!\top}YU_G$: $O(n^2r+nr^2)$), inner products and scatter-adds over $q$ terms ($O(qr)$), eigendecompositions of $n\times n$ and $r\times r$ symmetric matrices ($O(n^3)$, $O(r^3)$), and Cholesky factorization/solve of an SPD matrix of order $nr$ ($O((nr)^3)$). Forming $U^{\!\top}U=\sum_\ell u_\ell u_\ell^{\!\top}$ with $u_\ell\in\bR^{nr}$ is $q$ rank-one updates of an $nr\times nr$ matrix, i.e.\ $O(q(nr)^2)$. Summing per-step costs over $m$ iterations and adding setup gives (e). Every loop is over $\ell\in\{1,\dots,q\}$, and $\tilde Z\in\bR^{q\times r}$, the index arrays, and the values $(t_\ell)$ are the only $q$-sized data; $Z\in\bR^{M\times r}$ itself, $T\in\bR^{n\times M}$, $S\in\bR^{N\times q}$, and the intermediate $Y=KWZ^{\!\top}\in\bR^{n\times M}$ are never materialized (Lemmas~\ref{lem:matvec}--\ref{lem:form}). Hence no order-$M$ or order-$N$ work occurs.
\end{proof}

\paragraph{Why this is faster.} A naive solve of \eqref{eq:system} costs $O(n^3r^3)$ \emph{after} an order-$N$ or order-$M$ assembly of $A$. Algorithm~\ref{alg:pcg} removes the order-$N/M$ assembly entirely (Lemmas~\ref{lem:matvec}--\ref{lem:rhs}) and replaces the $O(n^3r^3)$ dense solve by $m=O(\log(1/\tau))$ iterations each costing $O(n^2r+nr^2+qr)$, plus a one-time $O(n^3+r^3)$ setup. Since $\varkappa(P^{-1}A)=O(1)$, this is $O\!\big(n^3+r^3+\log(1/\tau)\,(n^2r+nr^2+qr)\big)$, an asymptotic improvement over $O(n^3r^3)$ whenever $r\ge2$ (and a large practical improvement in all regimes), while strictly obeying the ``no order-$M$, no order-$N$'' constraint.
